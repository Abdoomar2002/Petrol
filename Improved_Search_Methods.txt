## CURRENT PROBLEMATIC SEARCH:
```csharp
// ❌ INEFFICIENT - Loads ALL data into memory then filters
public IEnumerable<T> Search(string term)
{
    var list = _context.Set<T>().AsNoTracking()
        .AsEnumerable() // ❌ LOADS ALL DATA INTO MEMORY
        .Where(entity => // ❌ THEN FILTERS IN MEMORY
        {
            var properties = typeof(T).GetProperties()
                .Where(p => p.CanRead && p.GetMethod.IsPublic && p.Name != "Password");
            
            return properties.Any(prop =>
            {
                var value = prop.GetValue(entity);
                return value != null && value.ToString().Contains(term);
            });
        })
        .ToList();
    return list;
}
```

## IMPROVED SEARCH METHODS:

### METHOD 1: Database-Level Search (RECOMMENDED)
```csharp
public IEnumerable<T> Search(string term, int maxResults = 100)
{
    if (string.IsNullOrWhiteSpace(term))
        return _context.Set<T>().Take(maxResults).ToList();
    
    // Get searchable properties (string properties only)
    var searchableProperties = typeof(T).GetProperties()
        .Where(p => p.PropertyType == typeof(string) && 
                    p.CanRead && 
                    p.GetMethod.IsPublic && 
                    p.Name != "Password")
        .ToList();
    
    if (!searchableProperties.Any())
        return _context.Set<T>().Take(maxResults).ToList();
    
    // Build dynamic query
    var query = _context.Set<T>().AsQueryable();
    
    foreach (var prop in searchableProperties)
    {
        var parameter = Expression.Parameter(typeof(T), "x");
        var property = Expression.Property(parameter, prop);
        var containsMethod = typeof(string).GetMethod("Contains", new[] { typeof(string) });
        var containsCall = Expression.Call(property, containsMethod, Expression.Constant(term));
        var lambda = Expression.Lambda<Func<T, bool>>(containsCall, parameter);
        
        query = query.Where(lambda);
    }
    
    return query.Take(maxResults).ToList();
}
```

### METHOD 2: Full-Text Search (BEST PERFORMANCE)
```csharp
public IEnumerable<T> FullTextSearch(string term, int maxResults = 100)
{
    if (string.IsNullOrWhiteSpace(term))
        return _context.Set<T>().Take(maxResults).ToList();
    
    // Use SQL Server Full-Text Search
    return _context.Set<T>()
        .FromSqlRaw($"SELECT * FROM {typeof(T).Name}s WHERE CONTAINS(*, '{term}')")
        .Take(maxResults)
        .ToList();
}
```

### METHOD 3: Specific Entity Search Methods
```csharp
// For Employee
public IEnumerable<Employee> SearchEmployees(string term, int maxResults = 100)
{
    if (string.IsNullOrWhiteSpace(term))
        return _context.Employees.Take(maxResults).ToList();
    
    return _context.Employees
        .Where(e => e.Name.Contains(term) || 
                    e.FinanceNumber.Contains(term) || 
                    e.DepartmentName.Contains(term))
        .Take(maxResults)
        .ToList();
}

// For Training
public IEnumerable<Training> SearchTrainings(string term, int maxResults = 100)
{
    if (string.IsNullOrWhiteSpace(term))
        return _context.Training.Take(maxResults).ToList();
    
    return _context.Training
        .Include(t => t.Place)
        .Include(t => t.TrainingType)
        .Where(t => t.Name.Contains(term) || 
                    t.Place.Name.Contains(term) || 
                    t.TrainingType.Name.Contains(term))
        .Take(maxResults)
        .ToList();
}

// For Program
public IEnumerable<Program> SearchPrograms(string term, int maxResults = 100)
{
    if (string.IsNullOrWhiteSpace(term))
        return _context.Programs.Take(maxResults).ToList();
    
    return _context.Programs
        .Include(p => p.ProgramType)
        .Where(p => p.Name.Contains(term) || 
                    p.ProgramType.Type.Contains(term))
        .Take(maxResults)
        .ToList();
}
```

### METHOD 4: Advanced Search with Multiple Criteria
```csharp
public class SearchCriteria
{
    public string SearchTerm { get; set; }
    public int? DepartmentId { get; set; }
    public DateTime? FromDate { get; set; }
    public DateTime? ToDate { get; set; }
    public int MaxResults { get; set; } = 100;
}

public IEnumerable<Training> AdvancedTrainingSearch(SearchCriteria criteria)
{
    var query = _context.Training
        .Include(t => t.Place)
        .Include(t => t.TrainingType)
        .Include(t => t.Program)
        .AsQueryable();
    
    if (!string.IsNullOrWhiteSpace(criteria.SearchTerm))
    {
        query = query.Where(t => t.Name.Contains(criteria.SearchTerm) || 
                                 t.Place.Name.Contains(criteria.SearchTerm) || 
                                 t.TrainingType.Name.Contains(criteria.SearchTerm));
    }
    
    if (criteria.DepartmentId.HasValue)
    {
        query = query.Where(t => t.DepartmentName == criteria.DepartmentId.ToString());
    }
    
    if (criteria.FromDate.HasValue)
    {
        query = query.Where(t => t.From >= criteria.FromDate.Value);
    }
    
    if (criteria.ToDate.HasValue)
    {
        query = query.Where(t => t.To <= criteria.ToDate.Value);
    }
    
    return query.Take(criteria.MaxResults).ToList();
}
```

### METHOD 5: Cached Search (For Frequently Searched Data)
```csharp
private static readonly MemoryCache _searchCache = new MemoryCache(new MemoryCacheOptions
{
    SizeLimit = 1000
});

public IEnumerable<T> CachedSearch(string term, int maxResults = 100)
{
    var cacheKey = $"{typeof(T).Name}_{term}_{maxResults}";
    
    if (_searchCache.TryGetValue(cacheKey, out IEnumerable<T> cachedResult))
        return cachedResult;
    
    var result = Search(term, maxResults).ToList();
    
    var cacheOptions = new MemoryCacheEntryOptions()
        .SetSize(1)
        .SetSlidingExpiration(TimeSpan.FromMinutes(5));
    
    _searchCache.Set(cacheKey, result, cacheOptions);
    
    return result;
}
```
